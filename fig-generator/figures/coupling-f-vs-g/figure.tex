  \def\w{4.2}\def\h{0.75}
  % strip: #1 x-offset, #2 y, #3 success start, #4 success end (fractions).
  % Success = crosshatch, failure = paper (pilot segment styling).
  \newcommand{\strip}[4]{%
    \fill[paper] (#1,#2) rectangle ++(\w,\h);
    \fill[figure crosshatch] ({#1+#3*\w},#2) rectangle ({#1+#4*\w},{#2+\h});
    \draw[ink,line width=1.4pt] (#1,#2) rectangle ++(\w,\h);
    \draw[ink] ({#1+#3*\w},#2) -- ++(0,\h) ({#1+#4*\w},#2) -- ++(0,\h);
    \foreach \t in {0.25,0.75} \draw[ink] ({#1+\t*\w},#2) -- ++(0,-0.15);
  }
  % one row of the joint-law table; #3=1 highlights it as a solid plate
  \newcommand{\law}[4]{%
    \ifnum#3=1
      \node[figure solid,anchor=west,minimum width=2.5cm,minimum height=6mm,inner sep=2pt] at (10.0,#4) {\textbf{$#1$}\quad\textbf{$#2$}};
    \else
      \node[anchor=west,minimum width=2.5cm,minimum height=6mm,inner sep=2pt] at (10.0,#4) {$#1$\quad$#2$};
    \fi
  }
  \FigureMasthead{-0.8}{12.6}{3.55}{}{}
  \strip{0}{0}{0}{0.25}
  \strip{5}{0}{0}{0.75}
